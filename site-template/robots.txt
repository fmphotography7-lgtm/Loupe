# NOTE ON HOW THIS FILE WORKS: a crawler obeys ONLY the most specific
# group that names it, and does NOT inherit rules from "User-agent: *".
# So every Disallow has to be repeated in each group. Miss one and the
# named bot is allowed everywhere the wildcard group forbids.

# ------------------------------------------------------------------
# Everyone
# ------------------------------------------------------------------

Allow: /
# Client galleries are private work belonging to one couple. The cart
# means nothing to a crawler.
Disallow: /client.html
Disallow: /cart.html
Disallow: /client-galleries/
Disallow: /client-images/

# ------------------------------------------------------------------
# AI assistants — deliberately welcome, in one group so the Disallows
# above apply to them too.
#
# Three OpenAI bots do three different jobs:
#   OAI-SearchBot  builds the index ChatGPT search retrieves from.
#                  Blocked = never a candidate to be cited. The one
#                  that matters most for being found.
#   ChatGPT-User   fetches this site live when someone asks about it.
#                  Blocked = cannot be pulled into the answer someone
#                  is reading right now.
#   GPTBot         collects training data only. Nothing to do with
#                  citation either way.
#
# Being named in an answer is a referral, and what is being sold is a
# photograph, which cannot be quoted away.
# ------------------------------------------------------------------

User-agent: ChatGPT-User
User-agent: GPTBot
